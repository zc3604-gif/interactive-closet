let art;
let clock = 0;
let moving = true;
let mx = 0;
let my = 0;

const ZOOM = 1.8;
const SPEED = 0.35;
const CONTRAST = 1.4;

function setup() {
  createCanvas(700, 700, WEBGL);
  pixelDensity(1);
  noStroke();

  art = createShader(vertexCode, fragmentCode);
}

function draw() {
  if (moving) {
    clock += min(deltaTime, 50) * 0.001 * SPEED;
  }

  if (
    mouseX >= 0 && mouseX <= width &&
    mouseY >= 0 && mouseY <= height
  ) {
    mx = lerp(mx, map(mouseX, 0, width, -1, 1), 0.04);
    my = lerp(my, map(mouseY, 0, height, 1, -1), 0.04);
  }

  shader(art);

  art.setUniform("uResolution", [width, height]);
  art.setUniform("uMouse", [mx, my]);
  art.setUniform("uTime", clock);
  art.setUniform("uZoom", ZOOM);
  art.setUniform("uContrast", CONTRAST);

  plane(width, height);
}

function keyPressed() {
  if (key === " ") {
    moving = !moving;
    return false;
  }

  if (key === "s" || key === "S") {
    saveCanvas("silver-math-art", "png");
  }
}

vec2 multiplyComplex(vec2 a, vec2 b) {
  return vec2(
    a.x * b.x - a.y * b.y,
    a.x * b.y + a.y * b.x
  );
}

vec2 divideComplex(vec2 a, vec2 b) {
  float denominator = max(dot(b, b), 0.00001);
  return vec2(
    a.x * b.x + a.y * b.y,
    a.y * b.x - a.x * b.y
  ) / denominator;
}
